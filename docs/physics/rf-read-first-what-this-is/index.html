# VP keystone: the vacuum's collective stiffness is MEASURED, not assumed.
# Soft monodisperse spheres are compressed to jamming. NO physical constant
# (c, m_p, alpha, 633 nm) enters -- only N and a box. If phi self-organises
# to random-close-packing (~0.64) with contacts <z> -> 6, the stiffness is an
# emergent, measured number.  Requires numpy.  Runtime ~ seconds.
import numpy as np
rng = np.random.default_rng(7)

def pbc(d, L):
    return d - L*np.round(d/L)

def energy_force(pos, R, L):              # harmonic soft-sphere overlaps
    d = pbc(pos[:,None,:] - pos[None,:,:], L)
    r = np.sqrt((d*d).sum(2)); np.fill_diagonal(r, 1.0)
    ov = 2*R - r
    m = ov > 0; np.fill_diagonal(m, False)
    U = 0.5*np.sum(ov[m]**2)
    F = (np.where(m, ov/r, 0.0)[:,:,None]*d).sum(1)
    return U, F, int(m.sum()//2)

def fire(pos, R, L, steps=250):           # FIRE energy minimisation
    v = np.zeros_like(pos); dt, a, npos = 0.04, 0.1, 0
    for _ in range(steps):
        _, F, _ = energy_force(pos, R, L)
        pos = (pos + dt*v + 0.5*dt*dt*F) % L
        _, F2, _ = energy_force(pos, R, L)
        v += 0.5*dt*(F + F2)
        if (F2*v).sum() > 0:
            npos += 1
            fn = np.linalg.norm(F2) + 1e-12
            v = (1-a)*v + a*(F2/fn)*np.linalg.norm(v)
            if npos > 5: dt, a = min(dt*1.1, 0.15), a*0.99
        else:
            npos, dt, a, v = 0, dt*0.5, 0.1, v*0
    return pos

N, L = 120, 1.0
pos = rng.random((N, 3))*L
coef = N*(4/3)*np.pi/L**3                  # phi = coef * R^3
phi_jam = None
print("  phi     U/N        <z>")
for R in np.linspace(0.102, 0.116, 26):    # slowly compress (grow radius)
    pos = fire(pos, R, L)
    U, _, c = energy_force(pos, R, L)
    phi, z = coef*R**3, 2*c/N
    if 0.585 < phi < 0.70:
        print(f"  {phi:.3f}   {U/N:.2e}   {z:.2f}")
    if phi_jam is None and U/N > 1e-7 and z >= 5.4:
        phi_jam = phi
print(f"\njamming onset phi_jam = {phi_jam:.3f}   [RCP ~ 0.64]")
print("contacts <z> climb through the isostatic 6 ; constants used: 0")
